```powershell
$env:JAVA_HOME = 'C:\Program Files\Eclipse Adoptium\jdk-21.0.12.101-hotspot'
.\gradlew.bat '-PregressionTests' '-PaccelerationOnly=true' '-PcompatibilityModsDirectory=build/mod-fixture-occultism-1.224.4-20261019' '-PregressionDirectory=build/regression-occultism-new-investigation' runRegression --console=plain
```

